def square(x) {
  return x * x
}

def factorial(n) {
  if n <= 1 { return 1 }
  return n * factorial(n - 1)
}
print "  factorial(6) = " + factorial(6) + "\n"

// Функция с возвратом массива
def makeRange(start, end) {
  res = []
  for (i = start; i <= end; i = i + 1) {
    res = push(res, i)
  }
  return res
}
print "  makeRange(10, 15) = " + makeRange(10, 15) + "\n"

// 8. ООП: КЛАССЫ, КОНСТРУКТОРЫ, this, КОМПОЗИЦИЯ

print "\n 7. КЛАССЫ И ООП\n"
class Animal {
  name age
  Animal(n, a) {
    this.name = n
    this.age = a
  }
  getInfo() {
    return this.name + " (age: " + this.age + ")"
  }
  speak() {
    print "  [" + this.name + "] издает звук!\n"
  }
}

class Dog {
  animal breed
  Dog(name, age, b) {
    this.animal = new Animal(name, age) // Композиция
    this.breed = b
  }
  bark() {
    this.animal.speak()
    print "  Порода: " + this.breed + "\n"
  }
}

dog = new Dog("Бобик", 5, "Корги")
print "  Инфо: " + dog.animal.getInfo() + "\n"
dog.bark()
